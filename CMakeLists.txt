cmake_minimum_required(VERSION 3.16)
project(Occupancy_SLAM VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(GNUInstallDirs)
include(CTest)
find_package(Eigen3 3.4 REQUIRED NO_MODULE)
find_package(OpenMP REQUIRED)
set(LIBIGL_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/libigl/include" CACHE PATH "Directory containing igl/")
if(NOT EXISTS "${LIBIGL_INCLUDE_DIR}/igl/sparse.h")
  message(FATAL_ERROR "Install libigl headers or set LIBIGL_INCLUDE_DIR; see docs/PIPELINES.md")
endif()
add_library(occupancy_solver src/core/linear_solver.cpp)
target_include_directories(occupancy_solver PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(occupancy_solver PUBLIC Eigen3::Eigen)
add_library(occupancy_2d_core MyStruct.cpp SubFuncs.cpp FuncConvertObs.cpp src/optimization/optimizer2d.cpp src/optimization/normal2d.cpp)
target_include_directories(occupancy_2d_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}> $<INSTALL_INTERFACE:include/occupancy/legacy> PRIVATE ${LIBIGL_INCLUDE_DIR})
target_link_libraries(occupancy_2d_core PUBLIC occupancy_solver OpenMP::OpenMP_CXX)
add_library(occupancy_2d_io src/io/scan2d.cpp src/mapping/export2d.cpp src/apps/run2d.cpp)
target_include_directories(occupancy_2d_io PRIVATE cpp ${LIBIGL_INCLUDE_DIR})
target_link_libraries(occupancy_2d_io PUBLIC occupancy_2d_core)
add_executable(occupancy_slam_2d cpp/main_2d.cpp)
target_link_libraries(occupancy_slam_2d PRIVATE occupancy_2d_io)
option(BUILD_3D "Build sparse 3D optimizer and PCD adapter" ON)
if(BUILD_3D)
  find_package(PCL REQUIRED COMPONENTS common io filters)
  add_library(occupancy_3d_core src/optimization/optimizer3d.cpp)
  target_link_libraries(occupancy_3d_core PUBLIC occupancy_solver)
  add_library(occupancy_3d_io src/io/pcd.cpp src/mapping/export3d.cpp src/apps/run3d.cpp)
  target_include_directories(occupancy_3d_io PRIVATE cpp ${PCL_INCLUDE_DIRS})
  target_link_libraries(occupancy_3d_io PUBLIC occupancy_3d_core PRIVATE ${PCL_LIBRARIES})
  add_executable(occupancy_slam_3d cpp/main_3d.cpp)
  target_link_libraries(occupancy_slam_3d PRIVATE occupancy_3d_io)
  install(TARGETS occupancy_slam_3d RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
install(TARGETS occupancy_slam_2d RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(TARGETS occupancy_solver occupancy_2d_core EXPORT OccupancySLAMTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
if(BUILD_3D)
  install(TARGETS occupancy_3d_core EXPORT OccupancySLAMTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
endif()
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES MyStruct.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/occupancy/legacy)
install(DIRECTORY tools config ros2 DESTINATION ${CMAKE_INSTALL_DATADIR}/occupancy_slam USE_SOURCE_PERMISSIONS PATTERN "__pycache__" EXCLUDE)
if(BUILD_TESTING)
  add_executable(occupancy_core_tests tests/test_core.cpp)
  target_include_directories(occupancy_core_tests PRIVATE ${LIBIGL_INCLUDE_DIR})
  target_link_libraries(occupancy_core_tests PRIVATE occupancy_2d_core)
  add_test(NAME core_n_approximation_and_solver COMMAND occupancy_core_tests)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME scan_io COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_scan_io.py)
  add_test(NAME display_styles COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_display_styles.py)
  add_test(NAME quantitative_evaluation COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_evaluate_2d.py)
  add_test(NAME cartesian_endpoint_input COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_endpoint_input.py)
  add_test(NAME multiresolution_2d COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_multiresolution_2d.py)
  add_test(NAME simulation_geometry COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_simulation_geometry.py)
  set_tests_properties(scan_io display_styles quantitative_evaluation cartesian_endpoint_input simulation_geometry multiresolution_2d PROPERTIES ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
endif()

install(PROGRAMS tools/occupancy-slam DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES requirements.txt requirements-bag.txt requirements-matlab.txt requirements-test.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/occupancy_slam)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/OccupancySLAMConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/OccupancySLAMConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/OccupancySLAM)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/OccupancySLAMConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(EXPORT OccupancySLAMTargets NAMESPACE OccupancySLAM:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/OccupancySLAM)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/OccupancySLAMConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/OccupancySLAMConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/OccupancySLAM)
