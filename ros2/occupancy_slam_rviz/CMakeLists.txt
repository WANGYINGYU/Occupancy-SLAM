cmake_minimum_required(VERSION 3.16)
project(occupancy_slam_rviz)
find_package(ament_cmake REQUIRED)
find_package(rviz_default_plugins REQUIRED)
find_package(pluginlib REQUIRED)
add_library(occupancy_slam_grayscale_map SHARED src/grayscale_map.cpp)
target_compile_features(occupancy_slam_grayscale_map PUBLIC cxx_std_17)
ament_target_dependencies(occupancy_slam_grayscale_map rviz_default_plugins pluginlib)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS occupancy_slam_grayscale_map LIBRARY DESTINATION lib)
ament_package()
